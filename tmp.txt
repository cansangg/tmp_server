git init                                   # 1. 注入 Git 灵魂
git add .                                  # 2. 把文件放进暂存区
git commit -m "Release: Tetris" # 3. 极其专业地写下最终版备注
git branch -M main                         # 4. 统一主分支名
git remote add origin git@github.com:cansangg/Tetris.git  # 5. 绑定新家地址（换成你刚才复制的链接）
git push -u origin main                    # 6. 一飞冲天！
